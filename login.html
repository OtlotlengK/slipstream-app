async function handleSubmit() { const email = document.getElementById('email').value.trim(); const password = document.getElementById('password').value; const submitBtn = document.getElementById('submit-btn'); if (!email || !password) { showMessage('Please enter your email and password.', 'error'); return; } submitBtn.disabled = true; submitBtn.textContent = loginMode ? 'Logging in...' : 'Creating account...'; // -------------------------------------------------- // LOGIN // -------------------------------------------------- if (loginMode) { const { error } = await _supabase.auth.signInWithPassword({ email, password }); if (error) { showMessage(error.message, 'error'); submitBtn.disabled = false; submitBtn.textContent = 'Log In'; return; } window.location.href = 'index.html'; return; } // -------------------------------------------------- // SIGN UP WITH METADATA (Triggers backend profile creation) // -------------------------------------------------- const businessName = document.getElementById('business-name').value.trim(); const ownerName = document.getElementById('owner-name').value.trim(); if (!businessName || !ownerName) { showMessage('Please enter your business name and your name.', 'error'); submitBtn.disabled = false; submitBtn.textContent = 'Create Merchant Account'; return; } const { error } = await _supabase.auth.signUp({ email, password, options: { data: { business_name: businessName, owner_name: ownerName } } }); if (error) { showMessage(error.message, 'error'); submitBtn.disabled = false; submitBtn.textContent = 'Create Merchant Account'; return; } showMessage('Account created successfully! Please check your email to verify your account before logging in.', 'success'); submitBtn.disabled = false; submitBtn.textContent = 'Create Merchant Account'; loginMode = true; toggleMode(); }
